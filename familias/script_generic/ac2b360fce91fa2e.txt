function Reg_soft(){
		var page = baseURL();
		var form = $('#dataFrmReg');
		if (form[0].checkValidity() === true) {
			$.ajax({
				type: "POST",
				data: {
				'addData': $("#dataFrmReg").serializeArray(),
				'reg':1
			},
				dataType: "json",
				url: page+'/RegisterOnline',
				beforeSend: function () {
				$(".loaderAjax").show();
			},
				success: function(result){
					$(".loaderAjax").fadeOut("slow");
					if(result.statusCode==200){
						$("#register").modal('hide');
						location.reload();
					}else if(result.statusCode==402){
						editRec();
					}else{
						setToast('bg-danger',result.error);
					}	 
				}
			});
		}
		form.addClass('was-validated');	
	}
	
	
function updrootpass(){
		var page = baseURL();
		var rootPass = document.getElementById('root_pass').value;
		$.ajax({
			type: "POST",
			data:{'rootPass':rootPass},
			dataType: "json",
			url: page+'/updateRootPass',
			success: function(result){
				if(result.statusCode==200){
					
					location.reload();
				}else if(result.statusCode==402){
					editRec();
				}else{
					$('#alertBx').removeClass();
                	$('#alertBx').addClass('error');
					$('#alertBx').find('#alertMsg').html(result.error);
				}	 
			}
		});
	}
